from typing import Literal

from pydantic import BaseModel, Field


# ---------- POST /care/run 요청 ----------

class CareRunRequest(BaseModel):
    # 대시보드에서 선택한 가상 어르신 ID.
    # 실제 개인정보가 아니라 Mock 프로필을 가리켜야 함.
    profile_id: str

    # 시연할 상황. 프런트의 시나리오 선택 항목과 이름을 맞출 것.
    # TODO: 팀에서 사용할 시나리오 목록을 확정하기.
    scenario: Literal["normal", "heat", "induction", "fall"]

    # 장애 주입 목록. 예: ["vital"], ["iot", "llm"].
    # 빈 목록이면 장애를 주입하지 않는 정상 실행.
    # TODO: 프런트와 AutoHealing 담당자에게 이 형식이 맞는지 확인하기.
    fault: list[Literal["vital", "iot", "vision", "llm"]] = Field(
        default_factory=list
    )


# ---------- 조사 에이전트 결과 ----------

class VitalData(BaseModel):
    # 분당 심박수. 단위: bpm.
    # TODO: Mock 데이터 담당자와 필드명·범위를 확인하기.
    heart_rate: int | None = None

    # 수축기 혈압. 단위: mmHg.
    systolic_bp: int | None = None

    # 이완기 혈압. 단위: mmHg.
    diastolic_bp: int | None = None

    # 체온. 단위: 섭씨.
    temperature_c: float | None = None

    # 산소포화도. 단위: 퍼센트.
    spo2: int | None = None

    # 측정 시각. ISO 8601 형식 사용을 제안.
    # 예: "2026-10-08T14:30:00+09:00"
    measured_at: str | None = None


class IndoorEnvironment(BaseModel):
    # 실내 온도. 단위: 섭씨.
    temperature_c: float | None = None

    # 실내 상대습도. 단위: 퍼센트.
    humidity_percent: float | None = None


class DeviceStatus(BaseModel):
    # 에어컨 상태. 실제 제어가 아니라 Mock 상태.
    air_conditioner: Literal["on", "off", "unknown"] = "unknown"

    # 인덕션 상태. 실제 제어가 아니라 Mock 상태.
    induction: Literal["on", "off", "unknown"] = "unknown"


class CctvAnalysis(BaseModel):
    # CCTV Mock 이미지 위치 또는 식별자.
    # 이미지 파일 자체를 JSON에 넣을지는 별도 합의가 필요함.
    image_path: str | None = None

    # 비전 분석 결과를 한 문장으로 요약.
    # 진단이 아니라 "넘어짐 의심"처럼 관찰 결과로 표현할 것.
    summary: str | None = None

    # 분석 성공·미사용·실패 상태.
    status: Literal["analyzed", "unavailable", "failed"] = "unavailable"


class CareStatusResult(BaseModel):
    # 한 번의 전체 실행을 구분하는 ID.
    # /care/run 응답과 이후 /care/approve 요청에서 같은 값을 사용.
    run_id: str

    # 조회 대상인 가상 어르신 ID.
    profile_id: str

    # 실행에 사용한 시연 시나리오.
    scenario: str

    # 바이탈 조회 결과.
    # 조회 실패 시 None으로 둘지, 별도 상태 필드로 표현할지 팀 합의 필요.
    vital: VitalData | None = None

    # 실내 온·습도 조회 결과.
    indoor_environment: IndoorEnvironment | None = None

    # 에어컨·인덕션 등 기기 상태.
    devices: DeviceStatus | None = None

    # CCTV 분석 결과.
    cctv: CctvAnalysis | None = None

    # Profile Agent가 계산한 취약도.
    # 지병명 등 민감한 원문은 넣지 않음.
    vulnerability: Literal["high", "medium", "low"] | None = None

    # 데이터 출처. cache 사용 여부는 AutoHealing 담당자와 의미를 맞출 것.
    source: Literal["mock", "cache", "mixed"] = "mock"

    # 조회 중 발생한 오류나 복구 관련 짧은 요약.
    # 민감정보나 CCTV 이미지 내용 자체는 기록하지 않음.
    errors: list[str] = Field(default_factory=list)


# ---------- Emergency Agent 결과 ----------

class ProposedAction(BaseModel):
    # 승인 후 호출할 Mock 도구 종류.
    # 실제 기기 제어·알림·119 호출은 하지 않음.
    tool: Literal[
        "control_device",
        "notify_guardian",
        "call_119",
        "find_hospital",
    ]

    # 보호자 승인 화면에 보여줄 조치 설명.
    # 예: "에어컨 켜기", "119 연락 필요 안내"
    description: str

    # 도구에 전달할 인자.
    # 구체적인 키는 두나의 MCP 도구 규격과 합의해야 함.
    arguments: dict = Field(default_factory=dict)


class EmergencyResult(BaseModel):
    # 어떤 실행에서 나온 긴급 결과인지 연결하는 ID.
    run_id: str

    # Emergency Agent가 코드 규칙으로 결정한 위험 단계.
    # TODO: README 위험도 값과 정확히 일치하는지 확인하기.
    risk_level: Literal["normal", "caution", "warning", "critical"]

    # 위험 단계의 근거. 설명 가능한 짧은 문장 목록.
    reasons: list[str] = Field(default_factory=list)

    # 보호자 승인 대상으로 제안할 조치 목록.
    proposed_actions: list[ProposedAction] = Field(default_factory=list)

    # 승인 Gate를 거쳐야 하는지 표시.
    # 기기 제어·119 관련 조치는 승인 전 실행하지 않는 원칙.
    requires_approval: bool = True

    # CCTV 분석을 사용했다면 요약 문장 저장.
    cctv_summary: str | None = None


# ---------- POST /care/run 응답 ----------

class CareRunResponse(BaseModel):
    # 실행 ID. 승인 대기 후 /care/approve에서 다시 사용.
    run_id: str

    # /care/run의 현재 종료 상태.
    # 정상은 즉시 반환, 긴급 상황은 승인 대기로 반환할 수 있음.
    status: Literal[
        "normal",
        "pending_approval",
        "recovered",
        "failed",
    ]

    # 대시보드 상단에 한 줄로 보여줄 진행 상태.
    # 예: "바이탈 확인 완료 · 보호자 승인 대기"
    current_step: str

    # 조사 결과. 화면에 필요한 값을 보여주기 위한 선택 필드.
    care_status: CareStatusResult | None = None

    # 긴급 판단 결과. 승인이 필요할 때 포함.
    emergency: EmergencyResult | None = None

    # 정상·실패 등 상태를 보충하는 사용자용 안내.
    message: str | None = None


# ---------- POST /care/approve 요청 ----------

class CareApprovalRequest(BaseModel):
    # 승인 대기 중인 실행 ID.
    # 서버에 저장된 해당 실행의 EmergencyResult를 찾는 데 사용.
    run_id: str

    # 보호자의 선택.
    decision: Literal["approve", "reject"]


# ---------- 승인 후 Action 결과 ----------

class ActionResult(BaseModel):
    # 실행한 Mock 도구 이름.
    tool: str

    # 성공·실패·거절로 인한 미실행을 구분.
    status: Literal["success", "failed", "skipped"]

    # 화면에 보여줄 처리 결과.
    message: str

    # 재시도나 대체 경로가 있었다면 짧게 기록.
    recovery: str | None = None


class CareApprovalResponse(BaseModel):
    # 승인 대상 실행 ID.
    run_id: str

    # 승인 처리 후 최종 상태.
    status: Literal[
        "approved",
        "rejected",
        "completed",
        "failed",
    ]

    # 화면 상단의 현재 처리 단계 한 줄.
    current_step: str

    # 승인 후 실행한 Mock Action 결과 목록.
    # 거절이면 빈 목록일 수 있음.
    actions: list[ActionResult] = Field(default_factory=list)

    # 보호자 대시보드에 표시할 결과 요약.
    guardian_message: str | None = None

    # 어르신용으로 표시할 짧은 안내.
    elder_message: str | None = None
